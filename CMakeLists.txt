cmake_minimum_required(VERSION 3.16)
project(ansa C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Werror -Wpedantic)
include_directories(hal)
enable_testing()
add_library(ansa_null_backend STATIC backends/virtual/null/null_backend.c)
target_include_directories(ansa_null_backend PUBLIC hal backends/virtual/null)
add_executable(test_null_backend tests/unit/test_null_backend.c)
target_include_directories(test_null_backend PRIVATE hal backends/virtual/null)
target_link_libraries(test_null_backend PRIVATE ansa_null_backend)
add_test(NAME null_backend_stub_test COMMAND test_null_backend)
add_library(ansa_avhp_core STATIC avhp/core/memory.c avhp/core/interrupts.c avhp/core/clock.c avhp/core/boot.c)
target_include_directories(ansa_avhp_core PUBLIC avhp/core)
add_executable(test_avhp_core tests/avhp_core/test_avhp_core.c)
target_link_libraries(test_avhp_core PRIVATE ansa_avhp_core)
add_test(NAME avhp_core_test COMMAND test_avhp_core)

add_library(s09_sensor_hal STATIC s09-hal/s09_sensor_hal.c)
target_include_directories(s09_sensor_hal PUBLIC s09-hal Drivers/ICM42688/Inc Drivers/BMP388/Inc Drivers/HMC5883L/Inc hal)
add_executable(test_sensor_hal_shims tests/virtual/test_sensor_hal_shims.c)
target_include_directories(test_sensor_hal_shims PRIVATE s09-hal Drivers/ICM42688/Inc Drivers/BMP388/Inc Drivers/HMC5883L/Inc hal)
target_link_libraries(test_sensor_hal_shims PRIVATE s09_sensor_hal)
add_test(NAME sensor_hal_shim_compile_test COMMAND test_sensor_hal_shims)

add_library(ansa_virtual_backend STATIC backends/virtual/virtual_backend.c)
target_include_directories(ansa_virtual_backend PUBLIC hal backends/virtual)
target_link_libraries(ansa_virtual_backend PUBLIC m)
add_executable(test_virtual_backend tests/virtual/test_virtual_backend.c)
target_link_libraries(test_virtual_backend PRIVATE ansa_virtual_backend)
add_test(NAME virtual_backend_compliance COMMAND test_virtual_backend)
foreach(sensor_test icm42688 bmp388 hmc5883l)
    if(sensor_test STREQUAL "icm42688")
        set(sensor_dir "Drivers/ICM42688")
        set(sensor_src "Drivers/ICM42688/Src/icm42688.c")
    elseif(sensor_test STREQUAL "bmp388")
        set(sensor_dir "Drivers/BMP388")
        set(sensor_src "Drivers/BMP388/Src/bmp388.c")
    else()
        set(sensor_dir "Drivers/HMC5883L")
        set(sensor_src "Drivers/HMC5883L/Src/hmc5883l.c")
    endif()
    add_executable(test_${sensor_test} tests/qemu_${sensor_test}/main.c tests/qemu_${sensor_test}/mock_${sensor_test}_bus.c ${sensor_src})
    target_include_directories(test_${sensor_test} PRIVATE ${sensor_dir}/Inc tests/qemu_${sensor_test})
    target_link_libraries(test_${sensor_test} PRIVATE m)
    add_test(NAME ${sensor_test}_driver_test COMMAND test_${sensor_test})
endforeach()
